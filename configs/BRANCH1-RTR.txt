hostname BRANCH1-RTR

ip routing
!
interface FastEthernet0/0
 ip address 203.0.114.1 255.255.255.252
 ip nat outside
 no shutdown
!
interface FastEthernet1/0
 ip address 172.16.1.1 255.255.255.0
 ip nat inside
 no shutdown

 no shutdown
